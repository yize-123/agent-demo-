```java
package com.example.aigent.tools;

import com.example.aigent.model.Tool;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.ArrayList;
import java.util.List;
import java.util.stream.Stream;

@Slf4j
@Component
public class LSRepoTool implements Tool {

    @Override
    public String getName() {
        return "list_directory";
    }

    @Override
    public String getDescription() {
        return "列出指定目录下的文件和文件夹，用于探索项目结构";
    }

    @Override
    public JsonNode getParameters() {
        ObjectMapper mapper = new ObjectMapper();
        ObjectNode params = mapper.createObjectNode();
        params.put("type", "object");
        ObjectNode properties = mapper.createObjectNode();
        ObjectNode dirParam = mapper.createObjectNode();
        dirParam.put("type", "string");
        dirParam.put("description", "要列出的目录路径，默认为当前目录");
        properties.set("directory", dirParam);
        params.set("properties", properties);
        return params;
    }

    @Override
    public String execute(JsonNode arguments) {
        try {
            String directory = arguments.has("directory")
                    ? arguments.get("directory").asText()
                    : System.getProperty("user.dir");

            Path path = Paths.get(directory);
            List<String> items = new ArrayList<>();

            try (Stream<Path> stream = Files.list(path)) {
                stream.limit(50).forEach(p -> {
                    String type = Files.isDirectory(p) ? "[dir]  " : "[file] ";
                    items.add(type + p.getFileName());
                });
            }

            return "{\"directory\": \"" + directory + "\", \"items\": [" +
                    String.join(",", items.stream().map(s -> "\"" + s + "\"").toList()) +
                    "]}";
        } catch (Exception e) {
            log.error("list_directory 错误", e);
            return "{\"error\": \"" + e.getMessage() + "\"}";
        }
    }
}
```

```java
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.List;

@Slf4j
@Component
public class ReadFileTool implements Tool {

    @Override
    public String getName() {
        return "read_file";
    }

    @Override
    public String getDescription() {
        return "读取指定文件的内容，支持大文件分块读取";
    }

    @Override
    public JsonNode getParameters() {
        ObjectMapper mapper = new ObjectMapper();
        ObjectNode params = mapper.createObjectNode();
        params.put("type", "object");
        ObjectNode properties = mapper.createObjectNode();
        ObjectNode fileParam = mapper.createObjectNode();
        fileParam.put("type", "string");
        fileParam.put("description", "要读取的文件路径");
        properties.set("file_path", fileParam);
        params.set("properties", properties);
        params.putArray("required").add("file_path");
        return params;
    }

    @Override
    public String execute(JsonNode arguments) {
        try {
            String filePath = arguments.get("file_path").asText();
            Path path = Paths.get(filePath);

            List<String> lines = Files.readAllLines(path);
            int maxLines = Math.min(lines.size(), 200);
            String content = String.join("\n", lines.subList(0, maxLines));

            return "{\"file_path\": \"" + filePath + "\", \"content\": \"" +
                    content.replace("\"", "\\\"") + "\", \"truncated\": " +
                    (lines.size() > 200) + "}";
        } catch (Exception e) {
            log.error("read_file 错误", e);
            return "{\"error\": \"" + e.getMessage() + "\"}";
        }
    }
}
```

**2.1.3 TodoManagerTool.java** (待办管理工具)

```java
package com.example.aigent.tools;

import com.example.aigent.model.Tool;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.ConcurrentHashMap;

@Slf4j
@Component
public class TodoManagerTool implements Tool {

    private final ConcurrentHashMap<String, TodoItem> todoItems = new ConcurrentHashMap<>();

    @Data
    @Builder
    @NoArgsConstructor
    @AllArgsConstructor
    private static class TodoItem {
        private String id;
        private String title;
        private String description;
        private boolean completed;
        private long createdAt;
    }

    @Override
    public String getName() {
        return "manage_todo";
    }

    @Override
    public String getDescription() {
        return "管理待办事项，支持创建、查询、标记完成等操作";
    }

    @Override
    public JsonNode getParameters() {
        ObjectMapper mapper = new ObjectMapper();
        ObjectNode params = mapper.createObjectNode();
        params.put("type", "object");
        ObjectNode properties = mapper.createObjectNode();
        ObjectNode actionParam = mapper.createObjectNode();
        actionParam.put("type", "string");
        actionParam.put("description", "操作类型: create, list, get, complete, delete");
        properties.set("action", actionParam);
        ObjectNode titleParam = mapper.createObjectNode();
        titleParam.put("type", "string");
        titleParam.put("description", "待办事项标题 (create时使用)");
        properties.set("title", titleParam);
        ObjectNode descParam = mapper.createObjectNode();
        descParam.put("type", "string");
        descParam.put("description", "待办事项描述 (create时使用)");
        properties.set("description", descParam);
        ObjectNode idParam = mapper.createObjectNode();
        idParam.put("type", "string");
        idParam.put("description", "待办事项ID (get, complete, delete时使用)");
        properties.set("todo_id", idParam);
        params.set("properties", properties);
        params.putArray("required").add("action");
        return params;
    }

    @Override
    public String execute(JsonNode arguments) {
        try {
            String action = arguments.get("action").asText();
            ObjectMapper mapper = new ObjectMapper();

            switch (action) {
                case "create":
                    String id = "todo_" + System.currentTimeMillis();
                    TodoItem item = TodoItem.builder()
                            .id(id)
                            .title(arguments.has("title") ? arguments.get("title").asText() : "无标题")
                            .description(arguments.has("description") ? arguments.get("description").asText() : "")
                            .completed(false)
                            .createdAt(System.currentTimeMillis())
                            .build();
                    todoItems.put(id, item);
                    return "{\"action\": \"create\", \"todo_id\": \"" + id + "\"}";

                case "list":
                    List<TodoItem> list = new ArrayList<>(todoItems.values());
                    return "{\"action\": \"list\", \"count\": " + list.size() + ", \"items\": " +
                            mapper.writeValueAsString(list) + "}";

                case "get":
                    String getid = arguments.get("todo_id").asText();
                    TodoItem get = todoItems.get(getid);
                    return "{\"action\": \"get\", \"item\": " + mapper.writeValueAsString(get) + "}";

                case "complete":
                    String compId = arguments.get("todo_id").asText();
                    TodoItem compItem = todoItems.get(compId);
                    if (compItem != null) {
                        compItem.setCompleted(true);
                        return "{\"action\": \"complete\", \"todo_id\": \"" + compId + "\", \"success\": true}";
                    }
                    return "{\"action\": \"complete\", \"success\": false, \"error\": \"not found\"}";

                case "delete":
                    String delId = arguments.get("todo_id").asText();
                    todoItems.remove(delId);
                    return "{\"action\": \"delete\", \"todo_id\": \"" + delId + "\", \"success\": true}";

                default:
                    return "{\"error\": \"unknown action: " + action + "\"}";
            }
        } catch (Exception e) {
            log.error("manage_todo 错误", e);
            return "{\"error\": \"" + e.getMessage() + "\"}";
        }
    }
}
```

```java
package com.example.aigent.service;

import com.example.aigent.model.Tool;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ArrayNode;
import com.fasterxml.jackson.databind.node.ObjectNode;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import java.util.HashMap;
import java.util.List;
import java.util.Map;

@Slf4j
@Service
public class ToolRegistry {

    private final Map<String, Tool> tools = new HashMap<>();
    private final ObjectMapper objectMapper;

    public ToolRegistry(List<Tool> toolList, ObjectMapper objectMapper) {
        this.objectMapper = objectMapper;
        for (Tool tool : toolList) {
            tools.put(tool.getName(), tool);
            log.info("工具已注册: {}", tool.getName());
        }
    }

    public Tool getTool(String name) {
        return tools.get(name);
    }

    public ArrayNode getToolsAsJsonSchema() {
        ArrayNode arrayNode = objectMapper.createArrayNode();
        for (Tool tool : tools.values()) {
            ObjectNode toolNode = objectMapper.createObjectNode();
            toolNode.put("type", "function");

            ObjectNode functionNode = objectMapper.createObjectNode();
            functionNode.put("name", tool.getName());
            functionNode.put("description", tool.getDescription());
            functionNode.set("parameters", tool.getParameters());

            toolNode.set("function", functionNode);
            arrayNode.add(toolNode);
        }
        return arrayNode;
    }
}
```

**2.2.2 ConversationHistory.java** (对话历史管理)

```java
package com.example.aigent.service;

import com.example.aigent.model.Conversation;
import com.example.aigent.model.Message;
import org.springframework.stereotype.Service;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

@Service
public class ConversationHistory {

    private final Map<String, Conversation> conversations = new ConcurrentHashMap<>();

    public String createConversation() {
        String id = UUID.randomUUID().toString();
        conversations.put(id, Conversation.builder().conversationId(id).build());
        return id;
    }

    public void addMessage(String conversationId, Message message) {
        Conversation conv = conversations.computeIfAbsent(conversationId,
                id -> Conversation.builder().conversationId(id).build());
        conv.addMessage(message);
    }

    public List<Message> getMessages(String conversationId) {
        Conversation conv = conversations.get(conversationId);
        return conv != null ? conv.getMessages() : new ArrayList<>();
    }

    public void deleteConversation(String conversationId) {
        conversations.remove(conversationId);
    }
}
```

```java
    private final RestTemplate restTemplate;
    private final ObjectMapper objectMapper;

    public OpenAIService(ObjectMapper objectMapper) {
        this.objectMapper = objectMapper;
        this.restTemplate = new RestTemplate();
    }

    public String chatCompletion(List<Message> messages) {
        try {
            String url = baseUrl + "/v1/chat/completions";

            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.APPLICATION_JSON);
            headers.setBearerAuth(apiKey);

            ObjectNode requestBody = objectMapper.createObjectNode();
            requestBody.put("model", model);
            requestBody.put("temperature", temperature);

            ArrayNode messagesNode = requestBody.putArray("messages");
            for (Message msg : messages) {
                ObjectNode msgNode = objectMapper.createObjectNode();
                msgNode.put("role", msg.getRole());
                msgNode.put("content", msg.getContent());
                if (msg.getToolName() != null) {
                    msgNode.put("name", msg.getToolName());
                }
                if (msg.getToolCallId() != null) {
                    msgNode.put("tool_call_id", msg.getToolCallId());
                }
                messagesNode.add(msgNode);
            }

            HttpEntity<String> entity = new HttpEntity<>(requestBody.toString(), headers);
            ResponseEntity<String> response = restTemplate.exchange(
                    url, HttpMethod.POST, entity, String.class);

            JsonNode responseJson = objectMapper.readTree(response.getBody());
            return responseJson.path("choices").get(0).path("message").path("content").asText();

        } catch (Exception e) {
            log.error("OpenAI API 调用错误", e);
            return "抱歉，发生了错误: " + e.getMessage();
        }
    }

    public ToolCall chatCompletionWithTools(List<Message> messages, ArrayNode tools) {
        try {
            String url = baseUrl + "/v1/chat/completions";

            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.APPLICATION_JSON);
            headers.setBearerAuth(apiKey);

            ObjectNode requestBody = objectMapper.createObjectNode();
            requestBody.put("model", model);
            requestBody.put("temperature", temperature);

            ArrayNode messagesNode = requestBody.putArray("messages");
            for (Message msg : messages) {
                ObjectNode msgNode = objectMapper.createObjectNode();
                msgNode.put("role", msg.getRole());
                msgNode.put("content", msg.getContent());
                if (msg.getToolName() != null) {
                    msgNode.put("name", msg.getToolName());
                }
                if (msg.getToolCallId() != null) {
                    msgNode.put("tool_call_id", msg.getToolCallId());
                }
                messagesNode.add(msgNode);
            }

            requestBody.set("tools", tools);

            HttpEntity<String> entity = new HttpEntity<>(requestBody.toString(), headers);
            ResponseEntity<String> response = restTemplate.exchange(
                    url, HttpMethod.POST, entity, String.class);

            JsonNode responseJson = objectMapper.readTree(response.getBody());
            JsonNode choice = responseJson.path("choices").get(0);
            JsonNode message = choice.path("message");

            if (message.has("tool_calls") && !message.path("tool_calls").isEmpty()) {
                JsonNode toolCall = message.path("tool_calls").get(0);
                return ToolCall.builder()
                        .id(toolCall.path("id").asText())
                        .toolName(toolCall.path("function").path("name").asText())
                        .arguments(toolCall.path("function").path("arguments").asText())
                        .build();
            }

            return ToolCall.builder().build();

        } catch (Exception e) {
            log.error("OpenAI API 工具调用错误", e);
            return ToolCall.builder().build();
        }
    }
}
```

创建目录 `src/main/java/com/example/aigent/agent`，然后创建：

```java
package com.example.aigent.agent;

import com.example.aigent.model.AgentResponse;
import com.example.aigent.model.Message;
import com.example.aigent.model.Tool;
import com.example.aigent.model.ToolCall;
import com.example.aigent.service.ConversationHistory;
import com.example.aigent.service.OpenAIService;
import com.example.aigent.service.ToolRegistry;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;

@Slf4j
@Component
public class ReActAgent {

    private static final String REACT_SYSTEM_PROMPT =
            "你是一个AI Agent，具备使用工具的能力。当用户提问时，如果需要外部信息，可以调用相应工具。" +
                    "请按照推理 -> 行动 -> 观察的方式工作。";

    private final OpenAIService openAIService;
    private final ToolRegistry toolRegistry;
    private final ConversationHistory conversationHistory;
    private final ObjectMapper objectMapper;

    @Value("${app.agent.max-tool-calls:5}")
    private int maxToolCalls;

    public ReActAgent(OpenAIService openAIService, ToolRegistry toolRegistry,
                     ConversationHistory conversationHistory, ObjectMapper objectMapper) {
        this.openAIService = openAIService;
        this.toolRegistry = toolRegistry;
        this.conversationHistory = conversationHistory;
        this.objectMapper = objectMapper;
    }

    public AgentResponse execute(String conversationId, String userMessage) {
        long startTime = System.currentTimeMillis();

        if (conversationId == null || conversationId.isEmpty()) {
            conversationId = conversationHistory.createConversation();
        }

        List<String> thoughts = new ArrayList<>();
        List<String> toolsUsed = new ArrayList<>();

        conversationHistory.addMessage(conversationId, Message.user(userMessage));
        List<Message> messages = new ArrayList<>(conversationHistory.getMessages(conversationId));

        messages.add(0, Message.system(REACT_SYSTEM_PROMPT));

        int toolCallCount = 0;
        boolean continueLoop = true;

        while (continueLoop && toolCallCount < maxToolCalls) {
            ToolCall toolCall = openAIService.chatCompletionWithTools(
                    messages,
                    toolRegistry.getToolsAsJsonSchema()
            );

            if (!toolCall.isValid()) {
                continueLoop = false;
                break;
            }

            thoughts.add("调用工具: " + toolCall.getToolName());
            toolsUsed.add(toolCall.getToolName());

            messages.add(Message.assistant("调用工具: " + toolCall.getToolName()));

            String toolResult = executeTool(toolCall);
            messages.add(Message.tool(toolCall.getToolName(), toolResult, toolCall.getId()));
            conversationHistory.addMessage(conversationId, Message.assistant("调用工具: " + toolCall.getToolName()));
            conversationHistory.addMessage(conversationId, Message.tool(toolCall.getToolName(), toolResult, toolCall.getId()));

            toolCallCount++;
        }

        String finalAnswer = openAIService.chatCompletion(messages);
        conversationHistory.addMessage(conversationId, Message.assistant(finalAnswer));

        long responseTimeMs = System.currentTimeMillis() - startTime;

        return AgentResponse.success(conversationId, finalAnswer, thoughts, toolsUsed, responseTimeMs);
    }

    private String executeTool(ToolCall toolCall) {
        try {
            Tool tool = toolRegistry.getTool(toolCall.getToolName());
            if (tool == null) {
                return "{\"error\": \"tool not found: " + toolCall.getToolName() + "\"}";
            }
            JsonNode args = objectMapper.readTree(toolCall.getArguments());
            return tool.execute(args);
        } catch (Exception e) {
            log.error("执行工具错误", e);
            return "{\"error\": \"" + e.getMessage() + "\"}";
        }
    }
}
```

```javascript
let currentConversationId = null;

function addMessage(role, content) {
    const container = document.getElementById('chatContainer');
    const welcome = container.querySelector('.welcome-message');
    if (welcome) {
        welcome.remove();
    }

    const messageDiv = document.createElement('div');
    messageDiv.className = 'message ' + role;
    messageDiv.innerHTML = `<div class="message-content">${content.replace(/\n/g, '<br>')}</div>`;
    container.appendChild(messageDiv);
    container.scrollTop = container.scrollHeight;
}

async function sendMessage() {
    const input = document.getElementById('userInput');
    const message = input.value.trim();

    if (!message) {
        return;
    }

    input.value = '';
    addMessage('user', message);

    try {
        const response = await fetch('/api/agent/chat', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                message: message,
                conversationId: currentConversationId
            })
        });

        const data = await response.json();

        if (data.conversationId) {
            currentConversationId = data.conversationId;
        }

        let fullAnswer = data.answer;

        if (data.toolsUsed && data.toolsUsed.length > 0) {
            fullAnswer = '🛠️ 使用了工具: ' + data.toolsUsed.join(', ') + '\n\n' + fullAnswer;
        }

        addMessage('assistant', fullAnswer);

    } catch (error) {
        addMessage('assistant', '抱歉，发生了错误: ' + error.message);
    }
}
```
